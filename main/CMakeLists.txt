# Capture the git identity of the app repo and of the pax-gfx checkout into a
# generated header, so every benchmark result can name the exact code state it
# came from.
#
# A generated header rather than add_compile_definitions(): the definitions
# would carry embedded quotes into ESP-IDF's generated build-property file and
# trip its parser.
#
# The CMAKE_CONFIGURE_DEPENDS entries are the important part: they force a
# reconfigure whenever either repo's HEAD moves or its index changes, so the
# embedded hash cannot go stale and misattribute a result to code it was not
# built from.
function(bench_git_info dir prefix out_var)
	set(_hash "unknown")
	set(_branch "unknown")
	set(_dirty 0)

	execute_process(
		COMMAND git -C "${dir}" rev-parse --short=12 HEAD
		OUTPUT_VARIABLE _out OUTPUT_STRIP_TRAILING_WHITESPACE
		ERROR_QUIET RESULT_VARIABLE _rc)

	if(_rc EQUAL 0)
		set(_hash "${_out}")

		execute_process(
			COMMAND git -C "${dir}" rev-parse --abbrev-ref HEAD
			OUTPUT_VARIABLE _out OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
		set(_branch "${_out}")

		# Tracked modifications only: untracked files do not change the build.
		execute_process(
			COMMAND git -C "${dir}" status --porcelain --untracked-files=no
			OUTPUT_VARIABLE _out OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
		if(NOT "${_out}" STREQUAL "")
			set(_dirty 1)
		endif()

		if(EXISTS "${dir}/.git/HEAD")
			set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${dir}/.git/HEAD")
		endif()
		if(EXISTS "${dir}/.git/index")
			set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${dir}/.git/index")
		endif()
	endif()

	set(${out_var}
		"#define ${prefix}_GIT \"${_hash}\"\n#define ${prefix}_BRANCH \"${_branch}\"\n#define ${prefix}_DIRTY ${_dirty}\n"
		PARENT_SCOPE)

	message(STATUS "${prefix}: ${_hash} (${_branch}) dirty=${_dirty}")
endfunction()

bench_git_info("${CMAKE_SOURCE_DIR}" BENCH_APP BENCH_APP_DEFS)
bench_git_info("${CMAKE_SOURCE_DIR}/components/robotman2412__pax-gfx" BENCH_PAX BENCH_PAX_DEFS)

file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/bench_build_info.h"
	"// Generated by main/CMakeLists.txt. Do not edit.\n#pragma once\n\n${BENCH_APP_DEFS}\n${BENCH_PAX_DEFS}")

)
